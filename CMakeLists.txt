cmake_minimum_required(VERSION 3.16)
project(Triage VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost 1.65 REQUIRED)
add_library(triage INTERFACE)
add_library(Triage::triage ALIAS triage)
target_compile_features(triage INTERFACE cxx_std_17)
target_include_directories(triage INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(triage INTERFACE Boost::boost)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(triage INTERFACE -ffp-contract=off)
elseif(MSVC)
    target_compile_options(triage INTERFACE /fp:strict)
endif()
option(TRIAGE_BUILD_TESTS "Build independent geometric tests" ON)
option(TRIAGE_BUILD_BENCHMARKS "Build benchmark and corpus validator" ON)
option(TRIAGE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
function(triage_executable name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE Triage::triage)
    if(TRIAGE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${name} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
triage_executable(triage_example examples/basic.cpp)
if(TRIAGE_BUILD_TESTS)
    enable_testing()
    triage_executable(triage_test tests/test.cpp)
    add_test(NAME geometry COMMAND triage_test)
    add_test(NAME example COMMAND triage_example)
endif()
if(TRIAGE_BUILD_BENCHMARKS)
    triage_executable(triage_bench bench/benchmark.cpp)
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS triage EXPORT TriageTargets)
install(EXPORT TriageTargets NAMESPACE Triage:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Triage)
configure_package_config_file(cmake/TriageConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/TriageConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Triage)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/TriageConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/TriageConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/TriageConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Triage)
